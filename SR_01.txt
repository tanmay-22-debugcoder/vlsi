useIEEE.STD_LOGIC_1164.ALL;
entityshiftregis
Port(clk :inSTD_LOGIC;
rst :inSTD_LOGIC;
load:inSTD_LOGIC;--parallelloadenable
si :inSTD_LOGIC;--serialinput
pi :inSTD_LOGIC_VECTOR(3downto0);--parallelinput
mode:inSTD_LOGIC_VECTOR(1downto0);--modeselect
po :outSTD_LOGIC_VECTOR(3downto0);--paralleloutput
so :outSTD_LOGIC);--serialoutput
endshiftreg;
architectureBehavioralofshiftregis
signalreg:STD_LOGIC_VECTOR(3downto0);
begin
process(clk,rst)
begin
ifrst='1'then
reg<="0000";
elsifrising_edge(clk)then
ifload='1'then
reg<=pi;--synchronousparallelload
else
casemodeis
when"00"=>--hold(nooperation)
reg<=reg;
when"01"=>--shiftright:si->MSB
reg<=si&reg(3downto1);
when"10"=>--shiftleft :si->LSB
reg<=reg(2downto0)&si;
whenothers=>--"11":parallelload
reg<=pi;
endcase;
endif;
endif;
endprocess;
po<=reg;
so<=reg(0);--serialoutput(LSB)
endBehavioral;
